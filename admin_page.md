## 1. ?? Global Security Guard (middleware.ts)
This Next.js Edge middleware runs before any dashboard page or API endpoint loads. It validates the user session via your Google Cloud Auth token and checks it against your private environment variable.

import { NextResponse } from 'next/server';import type { NextRequest } from 'next/server';
export async function middleware(request: NextRequest) {
  const { pathname } = request.nextUrl;

  // Protect all routes starting with /admin or /api/admin
  if (pathname.startsWith('/admin') || pathname.startsWith('/api/admin')) {
    
    // 1. Retrieve the session token generated by your Google Auth setup
    const sessionToken = request.cookies.get('__session')?.value; 
    if (!sessionToken) {
      return NextResponse.redirect(new URL('/login', request.url));
    }

    try {
      // 2. Decode token to get user email (Adapt based on your exact Google Auth client wrapper)
      const userEmail = await getEmailFromToken(sessionToken); 

      // 3. Match against your hardcoded, private deployment email
      if (userEmail !== process.env.ADMIN_EMAIL) {
        return NextResponse.redirect(new URL('/404', request.url)); // Secretive 404
      }
    } catch (error) {
      return NextResponse.redirect(new URL('/login', request.url));
    }
  }

  return NextResponse.next();
}

## 2. ?? Supabase Row-Level Security (RLS) Policy
Run this SQL script inside your Supabase SQL Editor. This acts as your database-level firewall. Even if someone bypasses your frontend, the database will reject any updates or deletions unless the requester's authenticated email exactly matches yours.

-- Enable Row Level Security on your critical tablesALTER TABLE profiles ENABLE ROW LEVEL SECURITY;ALTER TABLE files ENABLE ROW LEVEL SECURITY;
-- Create policy allowing ONLY the sole admin to edit/delete user profilesCREATE POLICY admin_manage_profiles ON profilesFOR ALLTO authenticatedUSING (auth.jwt() ->> 'email' = current_setting('app.settings.admin_email', true))WITH CHECK (auth.jwt() ->> 'email' = current_setting('app.settings.admin_email', true));

## 3. ?? Synchronized Cloudinary & Supabase Cascade Deletion
This Next.js API endpoint handles file moderation. When you click delete, it purges the file from Cloudinary storage first, and then removes its corresponding metadata tracking row from Supabase.

import { NextResponse } from 'next/server';import { v2 as cloudinary } from 'cloudinary';import { createClient } from '@supabase/supabase-js';
// Configure Cloudinary Admin SDK
cloudinary.config({
  cloud_name: process.env.CLOUDINARY_CLOUD_NAME,
  api_key: process.env.CLOUDINARY_API_KEY,
  api_secret: process.env.CLOUDINARY_API_SECRET,
});
// Configure Supabase Service Role (Bypasses standard restrictions safely for backend admin use)const supabase = createClient(
  process.env.SUPABASE_URL!,
  process.env.SUPABASE_SERVICE_ROLE_KEY!
);
export async function POST(request: Request) {
  try {
    const { fileId, cloudinaryPublicId } = await request.json();

    if (!fileId || !cloudinaryPublicId) {
      return NextResponse.json({ error: 'Missing parameters' }, { status: 400 });
    }

    // Step 1: Destroy physical asset in Cloudinary storage bucket
    const cloudinaryResponse = await cloudinary.uploader.destroy(cloudinaryPublicId);
    
    if (cloudinaryResponse.result !== 'ok' && cloudinaryResponse.result !== 'not_found') {
      return NextResponse.json({ error: 'Cloudinary deletion failed' }, { status: 500 });
    }

    // Step 2: Remove tracking metadata row from Supabase database
    const { error: dbError } = await supabase
      .from('files')
      .delete()
      .eq('id', fileId);

    if (dbError) {
      return NextResponse.json({ error: dbError.message }, { status: 500 });
    }

    return NextResponse.json({ success: true, message: 'File purged everywhere successfully.' });
  } catch (err: any) {
    return NextResponse.json({ error: err.message }, { status: 500 });
  }
}
